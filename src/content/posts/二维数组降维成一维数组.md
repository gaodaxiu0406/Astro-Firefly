## 把`[[2,5],[7,3]]`降维成`[2,5,7,3]`
<!-- more -->

### 方法一:
```js
let arr=[[2,5],[7,3]];
arr.toString().split  //[2,5,7,3]
```

### 方法二:
```js
function fn(arr){
    let a=[];
    arr.map(item=>{
        a=[...a,...item]
    });
    return a
}
fn([[2,5],[7,3]])  //[2,5,7,3]
```

### 方法三:reduce
```js
let arr=[[1,2,3],[4,5,6],[7,8,9]].reduce(function(prev,next){
    return prev.concat(next)
})
console.log(arr)//[1,2,3,4,5,6,7,8,9]
```
